import html
import json
import re
import numpy as np
import pandas as pd

url_re = re.compile(r"https?://\S+|www\.\S+", re.I)
space_re = re.compile(r"\s+")
br_re = re.compile(r"(<br\s*/?>)+", re.I)


def to_str(x):
    if pd.isna(x):
        return None
    return str(x)


def clean_text(x):
    x = to_str(x)
    if x is None:
        return None
    x = html.unescape(x)
    x = br_re.sub(" ", x)
    x = x.replace("\r", " ").replace("\n", " ").replace("\t", " ")
    x = url_re.sub("[URL]", x)
    x = space_re.sub(" ", x).strip()
    return x if x else None


def word_count(x):
    return len(x.split()) if isinstance(x, str) else 0


def char_count(x):
    return len(x) if isinstance(x, str) else 0


def avg_word_len(x):
    toks = x.split() if isinstance(x, str) else []
    return float(np.mean([len(t) for t in toks])) if toks else 0.0


def upper_ratio(x):
    if not isinstance(x, str):
        return 0.0
    letters = [c for c in x if c.isalpha()]
    if not letters:
        return 0.0
    return sum(1 for c in letters if c.isupper()) / len(letters)


def add_text_cols(df, text_col="text"):
    df = df.copy()
    df["text_lower"] = df[text_col].str.lower()
    df["n_words"] = df[text_col].apply(word_count)
    df["n_chars"] = df[text_col].apply(char_count)
    df["avg_word_len"] = df[text_col].apply(avg_word_len)
    df["upper_ratio"] = df[text_col].apply(upper_ratio)
    df["exclaim_cnt"] = df[text_col].apply(lambda s: s.count("!") if isinstance(s, str) else 0)
    df["question_cnt"] = df[text_col].apply(lambda s: s.count("?") if isinstance(s, str) else 0)
    df["length_bin"] = pd.cut(
        df["n_words"],
        bins=cfg["length_bins"],
        labels=cfg["length_labels"],
        include_lowest=True,
        right=True
    ).astype(str)
    return df


def dedup_by_split_priority(df, split_col, text_col, priority_map):
    df = df.copy()
    df["_split_priority"] = df[split_col].map(priority_map).fillna(9999)
    df["_text_key"] = df[text_col].astype(str).str.strip().str.lower()
    df = df.sort_values(by=["_split_priority", "_text_key"], ascending=[True, True])
    df = df.drop_duplicates(subset=["_text_key"], keep="first").copy()
    df = df.drop(columns=["_split_priority", "_text_key"])
    return df.reset_index(drop=True)


def profile_table(df, name):
    rows = []
    for sp, part in df.groupby("split", dropna=False):
        rows.append({
            "dataset": name,
            "split": sp,
            "rows": int(len(part)),
            "unique_text": int(part["text"].nunique()),
            "dup_text": int(len(part) - part["text"].nunique()),
            "mean_words": float(part["n_words"].mean()) if len(part) else np.nan,
            "median_words": float(part["n_words"].median()) if len(part) else np.nan,
            "p25_words": float(part["n_words"].quantile(0.25)) if len(part) else np.nan,
            "p75_words": float(part["n_words"].quantile(0.75)) if len(part) else np.nan,
            "min_words": float(part["n_words"].min()) if len(part) else np.nan,
            "max_words": float(part["n_words"].max()) if len(part) else np.nan,
            "labels": json.dumps(part["label"].value_counts(dropna=False).to_dict(), ensure_ascii=False)
        })
    return pd.DataFrame(rows)

TOKEN_RE = re.compile(r"\b\w+\b")
SENT_RE = re.compile(r"[.!?]+")

NEGATIONS = {'not', 'no', 'never', 'none', 'nobody', 'nothing', 'neither', 'nor', "n't"}
INTENSIFIERS = {'very', 'really', 'extremely', 'totally', 'absolutely', 'highly', 'so', 'super'}
DOWNTONERS = {'slightly', 'somewhat', 'barely', 'hardly', 'partly', 'fairly', 'rather'}
POSITIVE_LEXICON = {'good', 'great', 'excellent', 'amazing', 'wonderful', 'love', 'liked', 'happy', 'joy', 'fun'}
NEGATIVE_LEXICON = {'bad', 'terrible', 'awful', 'hate', 'hated', 'sad', 'angry', 'fear', 'boring', 'worse'}
NEUTRAL_OTHER_LEXICON = {'maybe', 'perhaps', 'surprised', 'curious', 'realize', 'realized', 'guess', 'wonder'}

def tokenize(text):
    if not isinstance(text, str):
        return []
    return TOKEN_RE.findall(text)

def split_sentences(text):
    if not isinstance(text, str):
        return []
    parts = [p.strip() for p in SENT_RE.split(text) if p.strip()]
    return parts

def safe_div(a, b):
    return float(a) / float(b) if b else 0.0

def per_100_words(count, n_words):
    return safe_div(count * 100.0, n_words)

def type_token_ratio(tokens):
    return safe_div(len(set(tokens)), len(tokens))

def long_word_ratio(tokens, min_len=7):
    return safe_div(sum(len(t) >= min_len for t in tokens), len(tokens))

def short_word_ratio(tokens, max_len=3):
    return safe_div(sum(len(t) <= max_len for t in tokens), len(tokens))

def all_caps_token_ratio(text):
    toks = tokenize(text)
    if not toks:
        return 0.0
    return safe_div(sum(t.isupper() and len(t) > 1 for t in toks), len(toks))

def repeat_char_ratio(text):
    if not isinstance(text, str) or not text:
        return 0.0
    repeat_hits = len(re.findall(r'(.)\1{2,}', text))
    return safe_div(repeat_hits, len(text))

def punct_density(text):
    if not isinstance(text, str) or not text:
        return 0.0
    punct = sum(ch in string.punctuation for ch in text)
    return safe_div(punct, len(text))

def count_lexicon(tokens, lexicon):
    return sum(t in lexicon for t in tokens)

def emotion_entropy(counts):
    arr = np.array(counts, dtype=float)
    s = arr.sum()
    if s <= 0:
        return 0.0
    p = arr / s
    p = p[p > 0]
    return float(-(p * np.log2(p)).sum())

def cliffs_delta(x, y):
    x = np.asarray(pd.Series(x).dropna(), dtype=float)
    y = np.asarray(pd.Series(y).dropna(), dtype=float)
    if len(x) == 0 or len(y) == 0:
        return np.nan
    xy = np.subtract.outer(x, y)
    return float((np.sum(xy > 0) - np.sum(xy < 0)) / (len(x) * len(y)))

def probability_vector_discrete(x, y):
    sx = pd.Series(x).dropna()
    sy = pd.Series(y).dropna()
    levels = sorted(set(sx.tolist()) | set(sy.tolist()))
    if not levels:
        return np.array([1.0]), np.array([1.0])
    px = sx.value_counts().reindex(levels, fill_value=0).to_numpy(dtype=float)
    py = sy.value_counts().reindex(levels, fill_value=0).to_numpy(dtype=float)
    px = px / px.sum() if px.sum() else np.ones_like(px) / len(px)
    py = py / py.sum() if py.sum() else np.ones_like(py) / len(py)
    return px, py

def probability_vector_continuous(x, y, n_bins=30):
    sx = pd.Series(x).dropna().astype(float)
    sy = pd.Series(y).dropna().astype(float)
    if len(sx) == 0 or len(sy) == 0:
        return np.array([1.0]), np.array([1.0])
    global_min = float(min(sx.min(), sy.min()))
    global_max = float(max(sx.max(), sy.max()))
    if global_min == global_max:
        return np.array([1.0]), np.array([1.0])
    bins = np.linspace(global_min, global_max, n_bins + 1)
    px, _ = np.histogram(sx, bins=bins)
    py, _ = np.histogram(sy, bins=bins)
    px = px.astype(float)
    py = py.astype(float)
    px = px / px.sum() if px.sum() else np.ones_like(px) / len(px)
    py = py / py.sum() if py.sum() else np.ones_like(py) / len(py)
    return px, py

def compute_jsd(x, y, feature_type):
    if feature_type == 'continuous':
        px, py = probability_vector_continuous(x, y, n_bins=cfg2['jsd_continuous_bins'])
    else:
        px, py = probability_vector_discrete(x, y)
    return float(jensenshannon(px, py, base=2.0) ** 2)

def extract_features(df):
    out = df.copy()

    tokens_lower = out['text_lower'].fillna('').apply(tokenize)
    sentences = out['text'].fillna('').apply(split_sentences)

    out['sentence_count'] = sentences.apply(len)
    out['avg_sentence_len_words'] = [safe_div(len(toks), len(sents)) if len(sents) else 0.0 for toks, sents in zip(tokens_lower, sentences)]
    out['type_token_ratio'] = tokens_lower.apply(type_token_ratio)
    out['long_word_ratio'] = tokens_lower.apply(long_word_ratio)
    out['short_word_ratio'] = tokens_lower.apply(short_word_ratio)

    out['all_caps_token_ratio'] = out['text'].apply(all_caps_token_ratio)
    out['repeat_char_ratio'] = out['text'].apply(repeat_char_ratio)
    out['punct_density'] = out['text'].apply(punct_density)
    out['exclaim_per_100w'] = [per_100_words(c, n) for c, n in zip(out['exclaim_cnt'], out['n_words'])]
    out['question_per_100w'] = [per_100_words(c, n) for c, n in zip(out['question_cnt'], out['n_words'])]

    out['negation_cnt'] = tokens_lower.apply(lambda toks: count_lexicon(toks, NEGATIONS))
    out['negation_per_100w'] = [per_100_words(c, n) for c, n in zip(out['negation_cnt'], out['n_words'])]
    out['intensifier_cnt'] = tokens_lower.apply(lambda toks: count_lexicon(toks, INTENSIFIERS))
    out['intensifier_per_100w'] = [per_100_words(c, n) for c, n in zip(out['intensifier_cnt'], out['n_words'])]
    out['downtoner_cnt'] = tokens_lower.apply(lambda toks: count_lexicon(toks, DOWNTONERS))
    out['downtoner_per_100w'] = [per_100_words(c, n) for c, n in zip(out['downtoner_cnt'], out['n_words'])]

    pos_cnt = tokens_lower.apply(lambda toks: count_lexicon(toks, POSITIVE_LEXICON))
    neg_cnt = tokens_lower.apply(lambda toks: count_lexicon(toks, NEGATIVE_LEXICON))
    neu_cnt = tokens_lower.apply(lambda toks: count_lexicon(toks, NEUTRAL_OTHER_LEXICON))

    out['positive_lexicon_ratio'] = [safe_div(c, n) for c, n in zip(pos_cnt, out['n_words'])]
    out['negative_lexicon_ratio'] = [safe_div(c, n) for c, n in zip(neg_cnt, out['n_words'])]
    out['neutral_other_ratio'] = [safe_div(c, n) for c, n in zip(neu_cnt, out['n_words'])]
    out['emotion_entropy'] = [emotion_entropy([p, n, u]) for p, n, u in zip(pos_cnt, neg_cnt, neu_cnt)]

    return out

class TextDataset(Dataset):
    def __init__(self, texts, labels, tokenizer, max_length):
        self.texts = list(texts)
        self.labels = None if labels is None else list(labels)
        self.tokenizer = tokenizer
        self.max_length = max_length

    def __len__(self):
        return len(self.texts)

    def __getitem__(self, idx):
        enc = self.tokenizer(
            self.texts[idx],
            truncation=True,
            padding='max_length',
            max_length=self.max_length,
            return_tensors='pt'
        )
        item = {k: v.squeeze(0) for k, v in enc.items()}
        if self.labels is not None:
            item['labels'] = torch.tensor(int(self.labels[idx]), dtype=torch.long)
        return item

class FusionDataset(Dataset):
    def __init__(self, df, tokenizer, max_length, feature_cols):
        self.df = df.reset_index(drop=True).copy()
        self.tokenizer = tokenizer
        self.max_length = max_length
        self.feature_cols = list(feature_cols)

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        row = self.df.iloc[idx]

        enc = self.tokenizer(
            str(row['text']),
            truncation=True,
            padding='max_length',
            max_length=self.max_length,
            return_tensors='pt'
        )

        item = {k: v.squeeze(0) for k, v in enc.items()}

        feat = row[self.feature_cols].astype(float).to_numpy(dtype=np.float32)
        item['features'] = torch.tensor(feat, dtype=torch.float32)
        item['labels'] = torch.tensor(int(row['label']), dtype=torch.long)

        return item


class EarlyFusionModel(nn.Module):
    def __init__(self, model_name, num_extra_features, dropout=0.1):
        super().__init__()
        self.encoder = AutoModel.from_pretrained(model_name)
        hidden_size = self.encoder.config.hidden_size
        self.dropout = nn.Dropout(dropout)
        self.classifier = nn.Linear(hidden_size + num_extra_features, 2)

    def forward(self, input_ids=None, attention_mask=None, features=None, labels=None):
        outputs = self.encoder(input_ids=input_ids, attention_mask=attention_mask)

        if hasattr(outputs, 'last_hidden_state'):
            cls_vec = outputs.last_hidden_state[:, 0, :]
        else:
            cls_vec = outputs[0][:, 0, :]

        if features is None:
            raise ValueError('features is None in EarlyFusionModel.forward().')

        features = features.to(device=cls_vec.device, dtype=cls_vec.dtype)

        if features.dim() != 2:
            raise ValueError(f'features must be 2D, got shape {tuple(features.shape)}')

        if cls_vec.size(0) != features.size(0):
            raise ValueError(
                f'batch mismatch: cls batch={cls_vec.size(0)}, features batch={features.size(0)}'
            )

        merged = torch.cat([self.dropout(cls_vec), features], dim=1)
        logits = self.classifier(merged)

        loss = None
        if labels is not None:
            labels = labels.to(logits.device)
            loss = nn.CrossEntropyLoss()(logits, labels)

        return {'loss': loss, 'logits': logits}


def train_early_fusion_model(train_df, valid_df, run_name):
    tokenizer = AutoTokenizer.from_pretrained(cfg3['text_model_name'])

def predict_early_fusion(trainer, tokenizer, df):
    ds = FusionDataset(
        df,
        tokenizer,
        cfg3['text_max_length'],
        cfg3['stable_features']
    )
    pred_output = trainer.predict(ds)
    logits = pred_output.predictions
    probs = torch.softmax(torch.tensor(logits), dim=1)[:, 1].numpy()
    return probs

def compute_ecr_tables(text_df, fusion_df, dataset_name):
    merged = text_df[['id', 'text', 'label', 'pred_prob', 'pred_label']].copy().rename(columns={
        'pred_prob': 'text_only_prob',
        'pred_label': 'text_only_pred',
    })
    fused = fusion_df[['id', 'pred_prob', 'pred_label']].copy().rename(columns={
        'pred_prob': 'fusion_prob',
        'pred_label': 'fusion_pred',
    })
    merged = merged.merge(fused, on='id', how='left')

    base_wrong = merged['text_only_pred'].astype(int) != merged['label'].astype(int)
    fusion_correct = merged['fusion_pred'].astype(int) == merged['label'].astype(int)
    corrected = base_wrong & fusion_correct

    text_conf = np.maximum(merged['text_only_prob'].astype(float), 1.0 - merged['text_only_prob'].astype(float))
    high_conf_wrong = base_wrong & (text_conf >= cfg3['high_conf_threshold'])
    hc_corrected = high_conf_wrong & fusion_correct

    denom_ecr = int(base_wrong.sum())
    denom_hc = int(high_conf_wrong.sum())

    ecr = float(corrected.sum() / denom_ecr) if denom_ecr > 0 else np.nan
    hc_ecr = float(hc_corrected.sum() / denom_hc) if denom_hc > 0 else np.nan

    merged['was_text_only_wrong'] = base_wrong
    merged['was_corrected'] = corrected
    merged['text_only_conf'] = text_conf
    merged['was_high_confidence_error'] = high_conf_wrong
    merged['was_high_confidence_corrected'] = hc_corrected
    merged['dataset_name'] = dataset_name

    summary = {
        'dataset_name': dataset_name,
        'text_only_errors': denom_ecr,
        'corrected_errors': int(corrected.sum()),
        'ecr': ecr,
        'text_only_high_conf_errors': denom_hc,
        'high_conf_corrected_errors': int(hc_corrected.sum()),
        'hc_ecr': hc_ecr,
    }
    return merged, summary

            stage = rel.parts[1] if len(rel.parts) > 1 and rel.parts[0] in ['data', 'outputs'] else 'root'
            ext = f.suffix.lower()
            rows.append({
                'file_name': f.name,
                'file_path': str(f),
                'relative_path': str(rel),
                'folder': str(f.parent),
                'stage': stage,
                'extension': ext,
                'size_bytes': stat.st_size,
                'modified_time': datetime.fromtimestamp(stat.st_mtime).isoformat(timespec='seconds')
            })

inv.head(20)
